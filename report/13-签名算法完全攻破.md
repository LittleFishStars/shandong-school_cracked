## 一、最终结论（可直接使用的公式）

$$\boxed{\ \text{sign}=\text{Base64}\Big(\ \text{AES-256-CBC}_{\text{key}=\text{signKey}\|\mathbf{0}^{16},\ \text{IV}=\texttt{01234ABCDEF56789}}\big(\ T(\text{SHA256}_{\text{hex}}(json))\ \big)\ \Big)\ }$$

其中 **变换 T**（64 字符 hex 的首尾 8 字符换位）：

```
T(s) = s[-8:] + s[8:-8] + s[:8]        # s 为 64 字符小写 hex
```

**密钥表**：

| 环境 | `signKey`（前 16 字节，后补 16 个 `0x00`） | 触发条件 |
|---|---|---|
| **生产** | `09L8MJRMJKXW8TPF` | `env ∉ {9, 11}` |
| **测试** | `Z3MDTC9JATA1SSYB` | `env ∈ {9, 11}` |

**固定 IV**：`01234ABCDEF56789`

**输出**：64 字符明文 → PKCS#7 填充到 **80 字节** → AES-CBC → Base64 = **108 字符**

## 二、可运行的 Python 实现

```python
import base64, hashlib, json
from Crypto.Cipher import AES
from Crypto.Util.Padding import pad

IV = b"01234ABCDEF56789"

# env ∈ {9, 11} → 测试密钥；否则生产密钥
SIGN_KEY_PROD = b"09L8MJRMJKXW8TPF" + bytes(16)   # 32 字节（AES-256）
SIGN_KEY_TEST = b"Z3MDTC9JATA1SSYB" + bytes(16)


def sign(body, env=0):
    """
    body: dict（会被紧凑序列化）或 str
    env : 与 Java 层 HcjNative.signRequest(json, env) 的第二个参数一致
    """
    key = SIGN_KEY_TEST if env in (9, 11) else SIGN_KEY_PROD

    # ① 紧凑 JSON（无空格、不转义非 ASCII）
    text = (json.dumps(body, separators=(",", ":"), ensure_ascii=False)
            if isinstance(body, dict) else str(body))

    # ② SHA-256 → 64 字符小写 hex
    sha = hashlib.sha256(text.encode("utf-8")).hexdigest().lower()

    # ③ ★ 首尾 8 字符换位：hex[-8:] + hex[8:-8] + hex[:8]
    swapped = sha[-8:] + sha[8:-8] + sha[:8]

    # ④ AES-256-CBC（PKCS#7 填充）→ Base64
    ct = AES.new(key, AES.MODE_CBC, IV).encrypt(pad(swapped.encode(), 16))
    return base64.b64encode(ct).decode()
```

## 七、完整可用的实现（含所有已知密钥）

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""闪动校园 v8.7.7 协议签名 / 加密完整实现（纯 Python）"""
import base64
import hashlib
import json

from Crypto.Cipher import AES, DES
from Crypto.Util.Padding import pad, unpad

IV = b"01234ABCDEF56789"

# ── 签名密钥（env ∈ {9,11} 用测试密钥）──
SIGN_KEY_PROD = b"09L8MJRMJKXW8TPF" + bytes(16)
SIGN_KEY_TEST = b"Z3MDTC9JATA1SSYB" + bytes(16)

# ── 字段加密密钥 ──
FIELD_KEY = b"F44B0282BEA83557" + bytes(16)

# ── 其它 ──
SLIDER_KEY = b"Ukp3hmSe7BmMcgbE"
RUN_IMG_RECORD_KEY = "2V8BQ8MYXWU10Y9Z"
DES_KEYS = [b"c6bd901f", b"0ddf44be", b"950e5927", b"6460d8df",
            b"11fb887f", b"117e4900", b"bd113950", b"a99f6f97"]


def _aes_enc(data: bytes, key: bytes, iv: bytes = IV) -> str:
    ct = AES.new(key, AES.MODE_CBC, iv).encrypt(pad(data, 16))
    return base64.b64encode(ct).decode()


def _aes_dec(b64: str, key: bytes, iv: bytes = IV) -> bytes:
    return unpad(AES.new(key, AES.MODE_CBC, iv).decrypt(base64.b64decode(b64)), 16)


def sign(body, env: int = 0) -> str:
    """HTTP 请求头 sign（与 native signRequest 完全等价）"""
    key = SIGN_KEY_TEST if env in (9, 11) else SIGN_KEY_PROD
    text = (json.dumps(body, separators=(",", ":"), ensure_ascii=False)
            if isinstance(body, dict) else str(body))
    sha = hashlib.sha256(text.encode("utf-8")).hexdigest().lower()
    swapped = sha[-8:] + sha[8:-8] + sha[:8]
    return _aes_enc(swapped.encode(), key)


def encrypt_field(plain: str) -> str:
    """手机号 / 密码 / 姓名 / 学号 等敏感字段"""
    return _aes_enc(plain.encode("utf-8"), FIELD_KEY)


def decrypt_field(b64: str) -> str:
    return _aes_dec(b64, FIELD_KEY).decode("utf-8")


def encrypt_slider_point(raw_x: float, y: float = 15.0) -> str:
    """滑块验证码坐标（AES-128-ECB）"""
    fake_e = int(raw_x - 6.5) + 16.5
    raw = json.dumps({"x": fake_e, "y": int(y)}, separators=(",", ":")).encode()
    ct = AES.new(SLIDER_KEY, AES.MODE_ECB).encrypt(pad(raw, 16))
    return base64.b64encode(ct).decode()


def run_img_record(run_record_code: str, timestamp: str,
                   build_version: str = "26101017",
                   app_version: str = "8.7.7") -> str:
    """跑步防篡改结算值"""
    raw = f"{run_record_code}{build_version}{app_version}{RUN_IMG_RECORD_KEY}{timestamp}"
    return hashlib.md5(raw.encode()).hexdigest().lower()


def des_decrypt(b64: str, idx: int) -> str:
    """多媒体资源解密"""
    ct = DES.new(DES_KEYS[idx % len(DES_KEYS)], DES.MODE_ECB).decrypt(base64.b64decode(b64))
    return unpad(ct, 8).decode("utf-8")


if __name__ == "__main__":
    # 自检：与 native 预言机输出对比
    assert sign({"probe": 1}, 0).startswith("82nIn4Xz+WshQAVduVf2A5xSzQRSdFC2")
    assert sign({}, 0).startswith("tL/UJwaTYM8us8R+MEvjgJap2FqlDvj9Uhjq9")
    assert sign({"a": 2}, 0).startswith("SyhWzEJUJ9kp9W7NVLJEPVngDmExDq19")
    assert sign({"probe": 1}, 9).startswith("EGUiM3Qx/Ykt4r2G06QOH6+7qOl1XMGT")
    print("✓ 全部自检通过")
```
